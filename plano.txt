💥 O QUE FALTA (e é simples)
1. 🔍 Buscar deck pelo código
Endpoint:
GET /api/decks/share/{code}

Controller:
@GetMapping("/share/{code}")
public ResponseEntity<?> getSharedDeck(@PathVariable String code) {

    DeckShare share = deckShareRepository.findByShareCode(code)
        .orElseThrow(() -> new RuntimeException("Link inválido"));

    if (!share.getIsPublic()) {
        return ResponseEntity.status(403).body("Deck não é público");
    }

    Deck deck = share.getDeck();

    return ResponseEntity.ok(deck);
}

2. 📥 Importar / clonar deck
Endpoint:
POST /api/decks/share/{code}/import

Controller:
@PostMapping("/share/{code}/import")
public ResponseEntity<?> importDeck(
        @PathVariable String code,
        @AuthenticationPrincipal User user
) {

    DeckShare share = deckShareRepository.findByShareCode(code)
        .orElseThrow(() -> new RuntimeException("Link inválido"));

    if (!share.getAllowClone()) {
        return ResponseEntity.status(403).body("Clone não permitido");
    }

    Deck original = share.getDeck();

    Deck novo = new Deck();
    novo.setName(original.getName());
    novo.setDescription(original.getDescription());
    novo.setColor(original.getColor());
    novo.setUser(user);

    deckRepository.save(novo);

    for (Card c : original.getCards()) {
        Card clone = new Card();
        clone.setFrontText(c.getFrontText());
        clone.setBackText(c.getBackText());
        clone.setDeck(novo);
        cardRepository.save(clone);
    }

    return ResponseEntity.ok(novo);
}
